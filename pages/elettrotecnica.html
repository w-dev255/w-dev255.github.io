Elettrotecnica

L’elettrotecnica è, in senso rigoroso, la disciplina tecnica che studia e applica i fenomeni elettrici, con particolare attenzione alle correnti forti, ai circuiti elettrici, alle reti di distribuzione e alle macchine elettriche.

Tensione elettrica

La tensione elettrica è la differenza di potenziale elettrico tra due punti ed esprime quanta energia serve per spostare una carica elettrica da un punto all’altro.

#include <stdio.h>

#include <stdlib.h>

typedef struct {

 double X;

 double I;

 double R;

} Tension;

int main(int argc, char* argv[]) {

 if (argc == 3) {

  printf("The arguments are enough.\n");

 } else if (argc < 3) {

  printf("Insufficient arguments.\n");

  return 1;

 } else if (argc > 3) {

  printf("Too many arguments.\n");

  return 1;

 }

 Tension V = {0};

 char* p = NULL;

 V.I = strtod(argv[1], &p);

 if (*p != '\0') {

  printf("Invalid I.\n");

  return 1;

 } else {

  printf("Valid I.\n");

 }

 V.R = strtod(argv[2], &p);

 if (*p != '\0') {

  printf("Invalid R.\n");

  return 1;

 } else {

  printf("Valid R.\n");

 }

 V.X = V.I * V.R;

 printf("%f V\n", V.X);

 return 0;

}

Corrente elettrica

La corrente elettrica è il flusso ordinato di cariche elettriche che attraversa un conduttore in un intervallo di tempo.

  printf("Valid R.\n");

 I.R = strtod(argv[2], &p);

 if (*p != '\0') {

  printf("Invalid V.\n");

  return 1;

 } else {

  printf("Valid V.\n");

 printf("%f A");

Resistenza elettrica

La resistenza elettrica è la grandezza che misura quanto un materiale ostacola il passaggio della corrente elettrica.

#include <stdio.h>

#include <stdlib.h>

typedef struct {

 double V;

 double I;

 double X;

} Electric_resistance;

int main(int argc, char* argv[]) {

 if (argc == 3) {

  printf("The arguments are enough.\n");

 } else if (argc < 3) {

  printf("The arguments are insufficient.\n");

  return 1;

 } else if (argc > 3) {

  printf("Too many arguments.\n");

  return 1;

 }

 Electric_resistance R = {0};

 char* p = NULL;

 R.V = strtod(argv[1], &p);

 if (*p != '\0') {

  printf("Invalid V.\n");

  return 1;

 } else {

  printf("Valid V.\n");

 }

 R.I = strtod(argv[2], &p);

 if (*p != '\0') {

  printf("Invalid I.\n");

  return 1;

 } else {

  printf("Valid I.\n");

 }

 R.X = R.V / R.I;

 printf("%f Ω\n", R.X);

 return 0;

}
